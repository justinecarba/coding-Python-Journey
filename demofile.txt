        "Take care!"
    ]
}


def chatbot(message):
    message = message.lower().strip()

    # Dictionary responses
    if message in responses:
        return random.choice(responses[message])

    # Time
    elif "time" in message:
        current = datetime.datetime.now().strftime("%I:%M %p")
        return f"The current time is {current}."

    # Date
    elif "date" in message:
        today = datetime.datetime.now().strftime("%B %d, %Y")
        return f"Today's date is {today}."

    # Math
    elif message.startswith("calculate "):
        try:
            expression = message.replace("calculate ", "")
            answer = eval(expression)
            return f"The answer is {answer}"
        except:
            return "Sorry, I couldn't calculate that."

    # Name memory
    elif message.startswith("my name is"):
        name = message.replace("my name is", "").strip().title()
        return f"Nice to meet you, {name}! I'll remember your name during this chat."

    # Joke
    elif "joke" in message:
        jokes = [
            "Why do programmers prefer dark mode? Because light attracts bugs!",
            "Why did Python break up with Java? Because Java had too many classes!",
            "Debugging: Being the detective in a crime movie where you're also the criminal."
        ]
        return random.choice(jokes)

    # Favorite language
    elif "favorite programming language" in message:
        return "Definitely Python! It's simple and powerful."

    else:
        return "Sorry, I don't understand that. Can you ask something else?"
